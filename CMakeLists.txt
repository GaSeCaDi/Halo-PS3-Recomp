cmake_minimum_required(VERSION 3.20)
project(h3ps3_hello LANGUAGES C)
include(ps3-self)
add_executable(h3ps3_hello src/main.c)
target_compile_features(h3ps3_hello PRIVATE c_std_11)
target_compile_options(h3ps3_hello PRIVATE -Wall -Wextra)
target_link_libraries(h3ps3_hello PRIVATE sysutil rt lv2)
ps3_add_self(h3ps3_hello)

# Standalone first resource reader; the original homebrew target remains available.
add_executable(h3ps3_map_reader src/map_reader/main.c src/map_reader/map_header.c)
target_compile_features(h3ps3_map_reader PRIVATE c_std_11)
target_compile_options(h3ps3_map_reader PRIVATE -Wall -Wextra -Werror)
target_link_libraries(h3ps3_map_reader PRIVATE fs_stub sysutil rt lv2)
ps3_add_self(h3ps3_map_reader OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/dist/map-reader")

add_executable(h3ps3_tag_browser src/tag_browser/main.c src/cache/cache_index.c src/map_reader/map_header.c)

target_include_directories(h3ps3_tag_browser PRIVATE "$ENV{PS3DK}/portlibs/ppu/include")
target_link_directories(h3ps3_tag_browser PRIVATE "$ENV{PS3DK}/portlibs/ppu/lib")
target_link_libraries(h3ps3_tag_browser PRIVATE z fs_stub sysutil rt lv2)
ps3_add_self(h3ps3_tag_browser OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/dist/tag-browser")

target_compile_options(h3ps3_texture_viewer PRIVATE -Wall -Wextra -Werror)
